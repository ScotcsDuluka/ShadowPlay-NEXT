cmake_minimum_required(VERSION 3.24)
project(DulukaQsv VERSION 2.16.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(INSTALL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(INSTALL_DEV OFF CACHE BOOL "" FORCE)
set(BUILD_EXPERIMENTAL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory(packages/libvpl ${CMAKE_BINARY_DIR}/dispatcher)
if(MINGW)
  target_link_libraries(VPL PRIVATE dxgi d3d9 dxva2 setupapi cfgmgr32 advapi32)
  target_link_options(VPL PRIVATE -static-libgcc -static-libstdc++)
endif()
add_library(DulukaQsvNative SHARED qsv_bridge.cpp)
target_link_libraries(DulukaQsvNative PRIVATE VPL d3d11 dxgi)
target_compile_definitions(DulukaQsvNative PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN MFX_DEPRECATED_OFF)
set_target_properties(DulukaQsvNative PROPERTIES PREFIX "")
if(MINGW)
  target_link_options(DulukaQsvNative PRIVATE -static-libgcc -static-libstdc++)
endif()
